quick_start.py 
import pygeovision as pgv

client = pgv.PyGeoVision()

# Search 22+ satellite providers
results = client.search(
    bbox=(-0.15, 51.47, -0.10, 51.52),
    date_range=("2024-06-01", "2024-06-30"),
    providers=["planetary_computer"],
    cloud_cover_max=10,
)

# Download + post-process
downloads = client.download(
    results[:2], output_dir="./data/",
    post_process=["reproject:EPSG:4326", "cog"],
)

# GeoAI — segment buildings
client.geoai.segment.buildings(
    downloads[0].path,
    output_vector="buildings.geojson",
)

Code Examples

From satellite to insight

search_download.py
import pygeovision as pgv

client = pgv.PyGeoVision()

# Add credentials (stored securely in system keyring via PyGeoFetch)
client.add_credentials("usgs", username="user", password="pass")
client.add_credentials("planet", api_key="PL_KEY")
client.add_credentials("copernicus", client_id="id", client_secret="secret")

# Search across 22+ providers — returns STAC 1.0 compliant results
results = client.search(
    bbox=(-0.15, 51.47, -0.10, 51.52),
    date_range=("2024-06-01", "2024-06-30"),
    providers=["planetary_computer", "copernicus", "usgs"],
    cloud_cover_max=10,
    max_results=20,
)

for r in results:
    print(r)  # [planetary_computer] Sentinel-2B | 2024-06-26 | cloud=0%

# Download with full post-processing chain (delegates to PyGeoFetch)
downloads = client.download(
    results[:3],
    output_dir="./data/",
    parallel=4,
    post_process=["reproject:EPSG:4326", "compress:lzw", "cog"],
    verify_checksum=True,
)
geoai_inference.py
ga = client.geoai  # lazy-loads GeoAI on first access

# ── Segmentation ──────────────────────────────────
ga.segment.buildings("sentinel2.tif",
    output_vector="buildings.geojson", confidence_threshold=0.5)

ga.segment.solar_panels("aerial.tif", output_vector="solar.geojson")
ga.segment.water("s2.tif", band_order="sentinel2")
ga.segment.with_sam("aerial.tif")  # SAM auto-segment

# ── Detection ─────────────────────────────────────
ga.detect.cars("aerial.tif", output_path="cars.geojson")
ga.detect.ships("port.tif", output_path="ships.geojson")
ga.detect.grounded("aerial.tif", "swimming pools")

# ── Cloud masking ────────────────────────────────
ga.cloud.predict("sentinel2.tif", output_path="cloud_mask.tif")

# ── Foundation models ────────────────────────────
model = ga.prithvi.load("prithvi-eo-1.0-100M")
ga.prithvi.infer("hls_tile.tif", model, output_path="pred.tif")

# ── Captioning with Moondream VLM ────────────────
caption = ga.caption.caption("tile.tif")
answer  = ga.caption.query("tile.tif", "How many buildings?")
model_training.py
# Export training chips from imagery + labels
client.geoai.train.generate_chips(
    "sentinel2.tif", "labels.tif", "./chips/",
    chip_size=256, overlap=32,
)

# Train semantic segmentation
client.geoai.train.segmentation(
    "./chips/", "building_model.pth",
    num_classes=2, epochs=100, backbone="efficientnet-b4",
)

# Land cover with unified focal loss
client.geoai.train.segmentation_landcover(
    "./lc_chips/", "landcover.pth",
    num_classes=11, loss_fn="unified_focal",
)

# timm-backbone segmentation (1000+ backbones)
client.geoai.train.timm_segmentation(
    "./chips/", "timm_seg.pth",
    backbone="convnext_base", num_classes=5,
)

# ONNX export for deployment
client.geoai.onnx.export(model, "model.onnx",
    input_shape=(1, 4, 512, 512))
yaml_pipeline.py
# Programmatic pipeline builder
pipeline = (
    client.create_pipeline("weekly-sentinel2")
    .search(
        providers=["planetary_computer", "copernicus"],
        bbox=(-74.1, 40.6, -73.7, 40.9),
        date_range="last_7_days",
        cloud_cover="0-10",
    )
    .filter("data.cloud_cover < 5")
    .download(
        parallel=4, output="./raw/",
        post_process=["reproject:EPSG:4326", "cog"],
    )
    .export(destination="s3://my-bucket/")
    .set_schedule("0 6 * * 1")  # Every Monday 06:00 UTC
)
pipeline.run()

# End-to-end AI pipeline (data + inference)
result = client.pipeline(
    "building_footprints",
    bbox=(-0.15, 51.47, -0.10, 51.52),
    date="2024-06", output_dir="./results/",
)
print(result.output_path, result.stats)

Get Started

Up and running in minutes

01
📦 Install PyGeoVision
# Core + GeoAI stack
pip install "pygeovision[geoai]"

# Add rasterio + geopandas
pip install "pygeovision[geo]"

# Everything
pip install "pygeovision[all]"
02
🔐 Add Provider Credentials
# CLI
pygeovision data auth add usgs \
  --username USER --password PASS
pygeovision data auth add planet \
  --api-key YOUR_KEY

# Python
client.add_credentials("copernicus",
    client_id="id", client_secret="s")
03
🛰️ Search & Download
pygeovision data search \
  --bbox -0.15 51.47 -0.10 51.52 \
  --date 2024-06 \
  --providers planetary_computer \
  --cloud-max 10 \
  --output results.geojson

pygeovision data download \
  --from-search results.geojson \
  --output ./data/ --parallel 4 \
  --post-process reproject:EPSG:4326,cog
04
🤖 Run GeoAI
# Segment buildings
pygeovision ai segment buildings \
  --input sentinel2.tif \
  --output buildings.tif \
  --vector buildings.geojson

# End-to-end pipeline
pygeovision pipeline building_footprints \
  --bbox -0.15 51.47 -0.10 51.52 \
  --date 2024-06

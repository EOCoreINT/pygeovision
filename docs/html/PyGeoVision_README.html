What is PyGeoVision?

PyGeoVision is a production-ready geospatial AI platform that unifies the two most powerful open-source geospatial packages into a single, coherent Python API. It is the integration layer that makes them work together as one world-class system.

🛰️

PyGeoFetch — Data

Search & download from 22+ satellite providers via a single interface. Handles auth, YAML pipelines, caching, retries, and post-processing chains.

🤖

GeoAI — Intelligence

State-of-the-art AI for satellite imagery. Segmentation, detection, classification, change detection, foundation models, embeddings, and more.

🔗

PyGeoVision — Bridge

Seamless integration: PyGeoFetch data flows directly into GeoAI inference, training, and 10 end-to-end geospatial workflows. Zero code duplication.

# Core — PyGeoFetch integration
pip install pygeovision

# Full stack — adds GeoAI (PyTorch, Transformers, SMP, Leafmap)
pip install "pygeovision[geoai]"

# Add raster/vector processing (rasterio, geopandas)
pip install "pygeovision[geo]"

# Full installation
pip install "pygeovision[all]"
import pygeovision as pgv

client = pgv.PyGeoVision()

# ── Authentication (stored in system keyring via PyGeoFetch) ──
client.add_credentials("usgs", username="user", password="pass")
client.add_credentials("planet", api_key="PL_KEY")
client.add_credentials("copernicus", client_id="id", client_secret="secret")

# ── Search 22+ providers ──────────────────────────────────────
results = client.search(
    bbox=(-0.15, 51.47, -0.10, 51.52),
    date_range=("2024-06-01", "2024-06-30"),
    providers=["planetary_computer", "copernicus", "usgs"],
    cloud_cover_max=10,
)
print(f"Found {len(results)} scenes")

for r in results[:3]:
    print(r)
# → [planetary_computer] Sentinel-2C | 2024-06-16 | cloud=0% score=0.99
# → [copernicus]         Sentinel-1A | 2024-06-12 | cloud=N/A

# ── Download with post-processing ────────────────────────────
downloads = client.download(
    results[:3],
    output_dir="./data/",
    parallel=4,
    post_process=["unzip", "reproject:EPSG:4326", "compress:lzw", "cog"],
    verify_checksum=True,
)
ga = client.geoai  # lazy-loads GeoAI on first access

# ── Segmentation ─────────────────────────────────────────────
ga.segment.buildings("sentinel2.tif",
    output_path="buildings.tif", output_vector="buildings.geojson")

ga.segment.solar_panels("aerial.tif", output_vector="solar.geojson")
ga.segment.water("s2.tif", output_path="water.tif", band_order="sentinel2")
ga.segment.with_sam("aerial.tif")  # Segment Anything Model

# ── Detection ────────────────────────────────────────────────
ga.detect.cars("aerial.tif", output_path="cars.geojson")
ga.detect.ships("port.tif", output_path="ships.geojson")
ga.detect.grounded("aerial.tif", "swimming pools")

# ── Change detection ──────────────────────────────────────────
ga.change.detect("london_2020.tif", "london_2024.tif", output_path="changes.tif")

# ── Train a model ─────────────────────────────────────────────
ga.train.segmentation("./chips/", "model.pth", num_classes=5, epochs=100)

# ── Cloud masking ─────────────────────────────────────────────
ga.cloud.predict("sentinel2.tif", output_path="cloud_mask.tif")
# ── Programmatic pipeline builder ────────────────────────────
pipeline = (
    client.create_pipeline("weekly-sentinel2")
    .search(
        providers=["planetary_computer", "copernicus"],
        bbox=(-74.1, 40.6, -73.7, 40.9),
        date_range="last_7_days",
        cloud_cover="0-10",
    )
    .filter("data.cloud_cover < 5")
    .download(
        parallel=4, output="./raw/",
        post_process=["unzip", "reproject:EPSG:4326", "cog"],
    )
    .export(destination="s3://my-bucket/")
    .set_schedule("0 6 * * 1")  # Every Monday 06:00 UTC
)
pipeline.run()

# ── End-to-end AI pipeline ────────────────────────────────────
result = client.pipeline("building_footprints",
    bbox=(-0.15, 51.47, -0.10, 51.52), date="2024-06")
print(result.output_path, result.stats)

CLI Reference

# ── System ───────────────────────────────────────────────────
pygeovision status
pygeovision doctor

# ── Authentication ───────────────────────────────────────────
pygeovision data auth add usgs --username USER --password PASS
pygeovision data auth add planet --api-key YOUR_KEY
pygeovision data auth add copernicus --client-id ID --client-secret S
pygeovision data auth list
pygeovision data auth test planetary_computer

# ── Search ───────────────────────────────────────────────────
pygeovision data search --bbox -0.15 51.47 -0.10 51.52 \
    --date 2024-06 --providers planetary_computer,copernicus \
    --cloud-max 10 --format table --output results.geojson

# ── Download ─────────────────────────────────────────────────
pygeovision data download --from-search results.geojson \
    --output ./data/ --parallel 4 --verify-checksum \
    --post-process unzip,reproject:EPSG:4326,compress:lzw,cog

# ── AI Segmentation ──────────────────────────────────────────
pygeovision ai segment buildings \
    --input sentinel2.tif --output buildings.tif --vector buildings.geojson
pygeovision ai segment water --input s2.tif --output water.tif

# ── AI Detection ─────────────────────────────────────────────
pygeovision ai detect ships --input port.tif --output ships.geojson
pygeovision ai detect grounded --input aerial.tif \
    --prompt "swimming pools" --output pools.geojson

# ── AI Training ──────────────────────────────────────────────
pygeovision ai train segmentation --data ./chips/ \
    --output model.pth --num-classes 5 --epochs 100 --backbone efficientnet-b4
pygeovision ai train land-cover --data ./lc/ \
    --output lc.pth --num-classes 11 --loss-fn unified_focal

# ── Pipelines ────────────────────────────────────────────────
pygeovision pipeline building_footprints \
    --bbox -0.15 51.47 -0.10 51.52 --date 2024-06
pygeovision pipeline change_detection \
    --bbox -74.1 40.6 -73.7 40.9 --date-before 2020-01 --date-after 2024-01
pygeovision pipeline list

# ── YAML Pipeline ────────────────────────────────────────────
pygeovision data pipeline run weekly.yaml
pygeovision data pipeline validate weekly.yaml
pygeovision data pipeline schedule weekly.yaml --cron "0 6 * * 1"
pygeovision data pipeline list
pygeovision data pipeline history

# ── Models ───────────────────────────────────────────────────
pygeovision models list
pygeovision models info unet_resnet50
pygeovision models cache

# ── Providers ────────────────────────────────────────────────
pygeovision data providers --open-only
pygeovision data providers --sar
